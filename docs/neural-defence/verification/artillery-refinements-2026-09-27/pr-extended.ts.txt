if(w.tick%1200!==0)sample();writeFileSync('/tmp/fuse-safe-artillery-refined/pr-extended.world.json',encodeState(w));writeFileSync('/tmp/fuse-safe-artillery-refined/pr-extended.replay.json',JSON.stringify({initial:JSON.parse(initial),commands,ticks:w.tick,hash:hashState(w)}));writeFileSync('/tmp/fuse-safe-artillery-refined/pr-extended.samples.json',JSON.stringify(samples));console.log(JSON.stringify({final:true,seconds:w.tick/20,finished:w.finished,winner:w.winnerId,hash:hashState(w)}));
