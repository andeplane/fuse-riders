import {readFileSync} from 'node:fs';
import {createMatch,loadMap,step} from './engine/index.ts';
import {aiCommands,type AiStrategy} from './engine/ai.ts';
const cases: [string,AiStrategy,AiStrategy][] = [['close-quarters','pressure','siege'],['narrow-front','pressure','siege'],['close-quarters','balanced','relay'],['narrow-front','balanced','relay']];
for(const [map,a,b] of cases) for(const slot of [0,1]) {
 let world=createMatch(loadMap(JSON.parse(readFileSync(`/Users/anderhaf/projects/personal/slackgame/games/neural-defence/maps/${map}.json`,'utf8'))),{},[{id:'alpha',slot},{id:'beta',slot:1-slot}]);
 let rejected=0; const builds:Record<string,Record<string,number>>={alpha:{},beta:{}};
 while(!world.finished&&world.tick<18000){world=step(world,[...aiCommands(world,'alpha',a),...aiCommands(world,'beta',b)]);for(const o of world.outcomes){if(o.type==='rejected')rejected++; if(o.type==='constructed'){const kind=world.structures.find(s=>s.cell===o.cell)?.kind??'unknown';const owner=builds[o.playerId!];if(owner)owner[kind]=(owner[kind]??0)+1;}}}
 console.log(JSON.stringify({key:`${map}-${a}-${b}-${slot}`,result:world.finished?world.players.find(p=>p.alive)?.id??'draw':'timeout',seconds:world.tick/20,rejected,builds,players:world.players.map(p=>({id:p.id,...p.statistics}))}));
}
