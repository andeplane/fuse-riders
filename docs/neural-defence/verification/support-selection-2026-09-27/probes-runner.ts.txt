import {readFileSync} from 'node:fs';
import {createMatch,loadMap,step} from './engine/index.ts';
import {aiCommands,type AiStrategy} from './engine/ai.ts';
const pairs: [AiStrategy,AiStrategy][] = [['balanced','economy'],['balanced','relay'],['balanced','defensive'],['pressure','siege']];
const map=loadMap(JSON.parse(readFileSync('/Users/anderhaf/projects/personal/slackgame/games/neural-defence/maps/skirmish-24.json','utf8')));
for(const [a,b] of pairs) for(const slot of [0,1]) {
 let world=createMatch(map,{},[{id:'alpha',slot},{id:'beta',slot:1-slot}]);
 let rejected=0;
 while(!world.finished&&world.tick<18000){world=step(world,[...aiCommands(world,'alpha',a),...aiCommands(world,'beta',b)]);rejected+=world.outcomes.filter(o=>o.type==='rejected').length;}
 console.log(JSON.stringify({key:`skirmish-24-${a}-${b}-${slot}`,result:world.finished?world.players.find(p=>p.alive)?.id??'draw':'timeout',seconds:world.tick/20,rejected,players:world.players.map(p=>({id:p.id,...p.statistics}))}));
}
