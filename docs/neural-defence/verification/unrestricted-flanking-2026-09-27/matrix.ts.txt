import {readFileSync} from 'node:fs';
import {createMatch,loadMap,step,hashState} from './engine/index.ts';
import {aiCommands,AI_STRATEGIES} from './engine/ai.ts';
const maps=(process.argv[2]??'close-quarters,skirmish-24').split(',');
const completed=new Set<string>();
for(const file of ['default.jsonl','probes.jsonl']) {
 const content=readFileSync(new URL(file,import.meta.url),'utf8').trim();
 for(const line of content.split('\n').filter(Boolean)) completed.add(JSON.parse(line).key);
}
for(const id of maps){
 const map=loadMap(JSON.parse(readFileSync(`/Users/anderhaf/projects/personal/slackgame/games/neural-defence/maps/${id}.json`,'utf8')));
 for(let a=0;a<AI_STRATEGIES.length;a++) for(let b=a;b<AI_STRATEGIES.length;b++) for(const slot of [0,1]) {
 const pair=[AI_STRATEGIES[a]!,AI_STRATEGIES[b]!] as const;
 const key=`${id}-${pair.join('-')}-${slot}`;
 if(completed.has(key))continue;
 let world=createMatch(map,{},[{id:'alpha',slot},{id:'beta',slot:1-slot}]);
 let rejected=0;
 while(!world.finished&&world.tick<18000){world=step(world,[...aiCommands(world,'alpha',pair[0]),...aiCommands(world,'beta',pair[1])]);rejected+=world.outcomes.filter(o=>o.type==='rejected').length;}
 console.log(JSON.stringify({key,result:world.finished?world.winnerId??'draw':'timeout',seconds:world.tick/20,rejected,hash:hashState(world),players:world.players.map(p=>({id:p.id,...p.statistics}))}));
 }
}
