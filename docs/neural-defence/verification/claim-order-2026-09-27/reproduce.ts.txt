const cellKeys = new Set(["cell", "from", "to", "destination", "fromCell"]);
function normalize(value: Json, rotate: boolean, key = ""): Json {
  if (typeof value === "number") return rotate && cellKeys.has(key) ? rotateCell(value) : rotate && key === "slot" ? 1 - value : value;
  if (value === null || typeof value !== "object") return value;
  if (Array.isArray(value)) {
    const result = value.map((item) => normalize(item, rotate));
    return key === "outcomes" ? result.sort((a, b) => JSON.stringify(a).localeCompare(JSON.stringify(b))) : result;
  }
  // Both worlds use this same, exhaustively checked rotationally symmetric map.
  // Remove its absolute spawn labels rather than rotating the shared map twice.
  return Object.fromEntries(Object.keys(value).filter((field) => field !== "map").sort().map((field) => {
    const item = value[field]!;
    if (field === "priorities" || field === "miningRemainders") {
      assert.ok(item !== null && typeof item === "object" && !Array.isArray(item));
      return [field, Object.fromEntries(Object.entries(item).map(([cell, count]) => [rotate ? rotateCell(Number(cell)) : Number(cell), count]).sort((a, b) => Number(a[0]) - Number(b[0])))];
    }
    return [field, normalize(item, rotate, field)];
  }));
}
function differences(a: Json | undefined, b: Json | undefined, path = "", result: { path: string; a?: Json; b?: Json }[] = []) {
  if (result.length >= 12 || JSON.stringify(a) === JSON.stringify(b)) return result;
  if (a === null || b === null || typeof a !== "object" || typeof b !== "object") {
    result.push({ path, a, b }); return result;
  }
  const left: Record<string, Json> = Object.fromEntries(Object.entries(a)), right: Record<string, Json> = Object.fromEntries(Object.entries(b));
  for (const key of new Set([...Object.keys(left), ...Object.keys(right)])) differences(left[key], right[key], `${path}.${key}`, result);
  return result;
}
