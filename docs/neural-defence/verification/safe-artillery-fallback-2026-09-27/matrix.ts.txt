import {readFileSync,appendFileSync,existsSync} from 'node:fs';
import {createHash} from 'node:crypto';
import {createMatch,loadMap,step,hashState} from './engine/index.ts';
import {aiCommands,AI_STRATEGIES} from './engine/ai.ts';
const root='/tmp/fuse-safe-artillery-fallback', shard=Number(process.argv[2]);
if(![0,1].includes(shard))throw Error('Provide shard0 or1');
const manifest=JSON.parse(readFileSync(`${root}/manifest.json`,'utf8'));
for(const [file,hash] of Object.entries(manifest.engineFiles))if(createHash('sha256').update(readFileSync(`${root}/engine/${file}`)).digest('hex')!==hash)throw Error('Engine drift:'+file);
const read=(f:string)=>existsSync(f)?readFileSync(f,'utf8').trim().split('\n').filter(Boolean).map(l=>JSON.parse(l)):[];
const initial=read(`${root}/all-results.jsonl`),done=new Set(initial.map(r=>r.key));
if(initial.length!==25||done.size!==25)throw Error('Expected25unique initial cases');
const output=`${root}/matrix-${shard}.jsonl`;
for(const r of read(output)){if(done.has(r.key))throw Error('Duplicate result:'+r.key);done.add(r.key)}
const maps=['close-quarters','skirmish-24','open-front','narrow-front','lean-resources'];
const baseline=new Map(maps.flatMap(m=>read(`/Users/anderhaf/projects/personal/slackgame/docs/neural-defence/verification/unrestricted-flanking-2026-09-27/${m}-results.jsonl`)).map(r=>[r.key,r]));
let index=0;for(const id of maps){const map=loadMap(JSON.parse(readFileSync(`/Users/anderhaf/projects/personal/slackgame/games/neural-defence/maps/${id}.json`,'utf8')));for(let a=0;a<AI_STRATEGIES.length;a++)for(let b=a;b<AI_STRATEGIES.length;b++)for(const slot of[0,1]){const pair=[AI_STRATEGIES[a]!,AI_STRATEGIES[b]!]as const,key=`${id}-${pair.join('-')}-${slot}`;if(initial.some(r=>r.key===key))continue;const assigned=index++%2;if(assigned!==shard||done.has(key))continue;let w=createMatch(map,{},[{id:'alpha',slot},{id:'beta',slot:1-slot}]),rejected=0;while(!w.finished&&w.tick<18000){w=step(w,[...aiCommands(w,'alpha',pair[0]),...aiCommands(w,'beta',pair[1])]);rejected+=w.outcomes.filter(o=>o.type==='rejected').length}const result=w.finished?w.winnerId??'draw':'timeout',base=baseline.get(key);if(!base)throw Error('Missing baseline:'+key);const row={key,result,seconds:w.tick/20,hash:hashState(w),rejected,baseline:{result:base.result,seconds:base.seconds},newTimeout:result==='timeout'&&base.result!=='timeout',changedOutcome:result!==base.result,players:w.players.map(p=>({id:p.id,...p.statistics}))};appendFileSync(output,JSON.stringify(row)+'\n');console.log(JSON.stringify({key,result,seconds:row.seconds,newTimeout:row.newTimeout,changedOutcome:row.changedOutcome}));}}
console.log(JSON.stringify({complete:true,shard,totalAssigned:shard===0?93:92}));
