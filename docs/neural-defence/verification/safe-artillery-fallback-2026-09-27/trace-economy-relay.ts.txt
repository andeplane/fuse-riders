let w=createMatch(loadMap(JSON.parse(readFileSync('/Users/anderhaf/projects/personal/slackgame/games/neural-defence/maps/narrow-front.json','utf8'))),{},[{id:'alpha',slot:0},{id:'beta',slot:1}]);const initial=encodeState(w),commands=[],events=[];let first=false,prior60;
function summary(w){return w.players.map(p=>{const own=w.structures.filter(s=>s.ownerId===p.id&&s.connected),enemies=w.structures.filter(s=>s.ownerId!==p.id),attackable=[...enemies.map(s=>s.cell),...w.players.filter(o=>o.id!==p.id).flatMap(o=>o.queue.filter(j=>j.paid&&j.upgradeFrom===undefined).map(j=>j.cell))];const threats=c=>enemies.filter(s=>s.connected&&canAttack(s.kind)&&attackCells(w.map,s.cell,s.kind).has(c));const weapons=own.filter(s=>canAttack(s.kind)).map(s=>({cell:s.cell,kind:s.kind,priority:p.priorities[s.cell]??0,ammo:w.particles.filter(q=>q.ownerId===p.id&&q.mode==='stationed'&&q.cell===s.cell&&q.destination===s.cell).length,targets:attackable.filter(c=>attackCells(w.map,s.cell,s.kind).has(c)),activeProtection:[...protectionCells(w,s.kind,s.cell)].filter(c=>(own.some(t=>t.cell===c)||p.queue.some(j=>j.paid&&j.cell===c))&&threats(c).length)}));return{id:p.id,statistics:p.statistics,biomass:p.biomass,research:p.research,queue:p.queue,worker:p.worker,priorities:p.priorities,connected:own.reduce((a,s)=>(a[s.kind]=(a[s.kind]??0)+1,a),{}),firingCount:weapons.filter(s=>s.targets.length).length,fightingCount:weapons.filter(s=>s.targets.length||s.activeProtection.length).length,weapons:weapons.filter(s=>s.targets.length||s.activeProtection.length||s.priority)}})}
while(!w.finished&&w.tick<18000){if(w.tick===16800)prior60=w.players.map(p=>({...p.statistics}));const cs=[...aiCommands(w,'alpha','economy'),...aiCommands(w,'beta','relay')],original=[...productionAi(w,'alpha','economy'),...productionAi(w,'beta','relay')];if(!first&&JSON.stringify(cs)!==JSON.stringify(original)){first=true;writeFileSync('/tmp/fuse-safe-artillery-fallback/er-first.world.json',encodeState(w));const entry={tick:w.tick,seconds:w.tick/20,candidate:cs,production:original,state:summary(w)};writeFileSync('/tmp/fuse-safe-artillery-fallback/er-first.json',JSON.stringify(entry,null,2));console.log(JSON.stringify({firstDifference:w.tick,seconds:w.tick/20,candidate:cs,production:original,fighting:entry.state.map(s=>({id:s.id,firing:s.firingCount,fighting:s.fightingCount}))}))}if(cs.length)commands.push({tick:w.tick,commands:cs});if(w.tick>=16800)events.push(...cs.map(c=>({tick:w.tick,...c})));w=step(w,cs);if(w.tick>16800)events.push(...w.outcomes.filter(o=>!['income','dispatched'].includes(o.type)))}const hash=hashState(w);writeFileSync('/tmp/fuse-safe-artillery-fallback/er-final.world.json',encodeState(w));writeFileSync('/tmp/fuse-safe-artillery-fallback/er.replay.json',JSON.stringify({initial:JSON.parse(initial),commands,ticks:w.tick,hash}));writeFileSync('/tmp/fuse-safe-artillery-fallback/er-last60.events.json',JSON.stringify(events));const report={tick:w.tick,finished:w.finished,hash,state:summary(w),last60:w.players.map((p,i)=>({id:p.id,...Object.fromEntries(Object.entries(p.statistics).map(([k,v])=>[k,v-prior60[i][k]]))}))};writeFileSync('/tmp/fuse-safe-artillery-fallback/er-final.json',JSON.stringify(report,null,2));console.log(JSON.stringify({finalTick:w.tick,finished:w.finished,hash,last60:report.last60}));
